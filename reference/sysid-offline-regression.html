线性回归目标（固定 URDF 惯性）

τmeas = τdyn,URDF(q,q̇,q̈) + Yfric · πfric

τres = τmeas − τdyn,URDF = Yfric · πfric

πfric = [ Fv₁…Fv₄ ; Fc₁…Fc₄ ]

Yfric = [ diag(q̇) | diag(sign(q̇)) ] → (4 × 8)

Pinocchio 调用

τ_dyn = pin.rnea(model, data, q, v, a)[arm_indices]
τ_res = τ_meas - τ_dyn
Y_fric = [diag(q̇_arm), diag(sign(q̇_arm))]
π̂_fric = lstsq(Y_stack, τ_res_stack)

合成数据（Issue 02）

τ_meas = rnea[arm] + Fv·q̇ + Fc·sign(q̇) + noise
τ_res = τ_meas - rnea[arm]
π̂ = lstsq(Y_fric_stack, τ_res_stack)   # 8 unknowns
residual = ||Y_fric·π̂ − τ_res||
